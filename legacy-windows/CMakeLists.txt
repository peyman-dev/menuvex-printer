cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(MenuVexLegacy VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
# Immutable dependency revisions; never download a floating branch in a release build.
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03)
FetchContent_Declare(sqlite_source GIT_REPOSITORY https://github.com/SRombauts/SQLiteCpp.git GIT_TAG 3c31070f0d1b1c46ce4b5ae48bdd0279e3610fa4)
FetchContent_GetProperties(json)
if(NOT json_POPULATED)
  FetchContent_Populate(json)
endif()
FetchContent_GetProperties(sqlite_source)
if(NOT sqlite_source_POPULATED)
  FetchContent_Populate(sqlite_source)
endif()
add_library(legacy_sqlite STATIC ${sqlite_source_SOURCE_DIR}/sqlite3/sqlite3.c)
target_include_directories(legacy_sqlite PUBLIC ${sqlite_source_SOURCE_DIR}/sqlite3)
target_compile_definitions(legacy_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=0)
add_library(legacy_core STATIC src/core.cpp)
target_include_directories(legacy_core PUBLIC src ${json_SOURCE_DIR}/single_include)
target_link_libraries(legacy_core PUBLIC legacy_sqlite)
if(UNIX)
  target_link_libraries(legacy_sqlite PUBLIC pthread dl)
endif()
if(WIN32)
  enable_language(RC)
  add_compile_definitions(UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601 NTDDI_VERSION=0x06010000 NOMINMAX)
  target_compile_definitions(legacy_sqlite PRIVATE WINVER=0x0601 _WIN32_WINNT=0x0601)
  FetchContent_Declare(websocketpp GIT_REPOSITORY https://github.com/zaphoyd/websocketpp.git GIT_TAG 0e4241727c199e208ceb41134e840ba9968cf181)
  FetchContent_Declare(asio GIT_REPOSITORY https://github.com/chriskohlhoff/asio.git GIT_TAG 42a93679dc4c8c5caf3d3082542f1bfa2438271d)
  FetchContent_Populate(websocketpp)
  FetchContent_Populate(asio)
  configure_file(src/resources.rc.in resources.rc @ONLY)
  add_executable(menuvex-printer-legacy WIN32 src/main.cpp src/platform.cpp src/server.cpp ${CMAKE_CURRENT_BINARY_DIR}/resources.rc)
  target_include_directories(menuvex-printer-legacy PRIVATE ${websocketpp_SOURCE_DIR} ${asio_SOURCE_DIR}/asio/include)
  target_compile_definitions(menuvex-printer-legacy PRIVATE ASIO_STANDALONE _WEBSOCKETPP_CPP11_STL_ ASIO_DISABLE_STD_ALIGNED_ALLOC)
  target_link_libraries(menuvex-printer-legacy PRIVATE legacy_core ws2_32 mswsock bcrypt crypt32 advapi32 shell32 ole32 user32 gdi32 winspool usp10 comctl32)
  if(MSVC)
    target_compile_options(legacy_core PRIVATE /utf-8 /W4)
    target_compile_options(menuvex-printer-legacy PRIVATE /utf-8 /W4)
    target_link_options(menuvex-printer-legacy PRIVATE /SUBSYSTEM:WINDOWS,6.01 /MANIFEST:NO)
  endif()
endif()
enable_testing()
add_executable(legacy_core_tests tests/core_tests.cpp)
target_link_libraries(legacy_core_tests PRIVATE legacy_core)
add_test(NAME legacy_core COMMAND legacy_core_tests)
if(WIN32)
  add_executable(legacy_windows_integration tests/windows_integration.cpp src/platform.cpp src/server.cpp ${CMAKE_CURRENT_BINARY_DIR}/resources.rc)
  target_include_directories(legacy_windows_integration PRIVATE ${websocketpp_SOURCE_DIR} ${asio_SOURCE_DIR}/asio/include)
  target_compile_definitions(legacy_windows_integration PRIVATE MENUVEX_TESTING ASIO_STANDALONE _WEBSOCKETPP_CPP11_STL_ ASIO_DISABLE_STD_ALIGNED_ALLOC)
  target_link_libraries(legacy_windows_integration PRIVATE legacy_core ws2_32 mswsock bcrypt crypt32 advapi32 shell32 ole32 user32 gdi32 winspool usp10 comctl32)
  if(MSVC)
    target_compile_options(legacy_windows_integration PRIVATE /utf-8)
    target_link_options(legacy_windows_integration PRIVATE /MANIFEST:NO)
    target_compile_options(legacy_core_tests PRIVATE /utf-8)
  endif()
  add_test(NAME legacy_windows_sdk COMMAND legacy_windows_integration)
  set_tests_properties(legacy_windows_sdk PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.." TIMEOUT 60)
endif()
